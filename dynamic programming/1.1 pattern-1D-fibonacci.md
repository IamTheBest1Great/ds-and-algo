## 🟢 Pattern 1 — Linear DP (1D)

### 1.1 Fibonacci Style
**When to use:** Problems where the answer depends on previous one or two values (e.g., climbing stairs, house robber, decode ways).  
**State meaning:** `dp[i]` = answer for first i elements / value at i.

```cpp
// Template
int fibStyle(int n) {
    if (n <= 1) return n;
    vector<int> dp(n + 1);
    dp[0] = 0; dp[1] = 1; // base cases
    for (int i = 2; i <= n; ++i) {
        dp[i] = dp[i-1] + dp[i-2]; // transition (may be more complex)
    }
    return dp[n];
}
// Space-optimized: use two variables instead of array
```

**Key insight:** Often the transition is `dp[i] = dp[i-1] + dp[i-2]` or `dp[i] = max(dp[i-1], dp[i-2] + value[i])` (House Robber).

class Solution {
public:
    int climbStairs(int n) {
        vector<int> ans(n);
        if(n>=1)
        {
            ans[0]=1;
        }
        if(n>=2)
        {
            ans[1]=2;
        }
        for(int i=2;i<n;i++)
        {
            ans[i]=ans[i-1]+ans[i-2];
        }
        return ans[n-1];
    }
};

class Solution {
public:
    int minCostClimbingStairs(vector<int>& cost) 
    {
        int n=cost.size();
        for(int i=2;i<n;i++)
        {
            cost[i]=cost[i]+min(cost[i-1],cost[i-2]);
        }
        return min(cost[n-1],cost[n-2]); 
    }
};

198. House Robber
https://leetcode.com/problems/house-robber/description/
https://leetcode.com/problems/house-robber/solutions/7597758/dp-1d-fibonacci-very-easy-solution-by-sh-7kv2/

class Solution {
public:
    int rob(vector<int>& nums) 
    {
        int n=nums.size();
        vector<int> ans(n);
        if(n>=1)
        {
            ans[0]=nums[0];
        }
        if(n>=2)
        {
            ans[1] = max(nums[0],nums[1]);
        }
        for(int i=2;i<n;i++)
        {
            ans[i] = max(nums[i]+ans[i-2],ans[i-1]);
        }  
        
        if(n==1)
        {
            return ans[0];
        }
        return max(ans[n-1],ans[n-2]);  
    }
};

213. House Robber II
https://leetcode.com/problems/house-robber-ii/description/
https://leetcode.com/problems/house-robber-ii/solutions/7598548/dp-1d-fibonacci-by-shubham_sk70-47rm/



class Solution {
public:
    int helper(vector<int> nums, int start, int end)
    {
        int n=nums.size();
        vector<int> ans(n);
        int i;
        if(n>=1)
        {
            ans[start] = nums[start];
        }
        if(n>=2)
        {
            ans[start+1]=max(nums[start],nums[start+1]);
        }
        for(i=start+2; i<=end;i++)
        {
            ans[i]=max(nums[i]+ans[i-2],ans[i-1]);
        }
        return max(ans[end], ans[end-1]);
    }
    int rob(vector<int>& nums) { 
        int n=nums.size();
        if(n==1)
        {
            return nums[0];
        }
        if(n==2)
        {
            return max(nums[0],nums[1]);
        }
        int x = helper(nums,0,n-2);
        int y = helper(nums,1,n-1);
        return max(x,y);
    }
};

    https://leetcode.com/problems/decode-ways/solutions/7599162/dp-1d-fibonacci-by-shubham_sk70-gtj5/

    class Solution {
public:
    int numDecodings(string s) 
    {
        int n=s.length();
        vector<int> dp(n+1);
        if(n==0)
        {
            return 0; 
        }
        if(s[0]=='0')
        {
            return 0;
        }
        dp[0]=1;
        dp[1]=1;
        for(int i=2;i<=n;i++)
        {
            int singleDigit=stoi(s.substr(i-1,1));
            int doubleDigit=stoi(s.substr(i-2,2));
            if(singleDigit>0)
            {
                dp[i]+=dp[i-1];
            }
            if(doubleDigit>=10 && doubleDigit<=26)
            {
                dp[i]+=dp[i-2];
            }
        }
        return dp[n];
    }

};
